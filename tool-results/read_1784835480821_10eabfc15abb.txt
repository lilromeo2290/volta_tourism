   605→// ---- TRIP PLAN GENERATOR ----
   606→export function generateTripPlan(days: number, budget: string, interests: string[], style: string): TripPlan {
   607→  const destinationPool = destinations.filter(
   608→    (d) => interests.length === 0 || interests.some((i) => d.category.toLowerCase().includes(i.toLowerCase()))
   609→  );
   610→
   611→  const dayPlans: DayPlan[] = [];
   612→  const usedDestinations = new Set<string>();
   613→
   614→  for (let i = 0; i < days; i++) {
   615→    const dayNum = i + 1;
   616→    const availableDest = destinationPool.filter((d) => !usedDestinations.has(d.id));
   617→    const dest = availableDest.length > 0
   618→      ? availableDest[i % availableDest.length]
   619→      : destinationPool[i % destinationPool.length];
   620→
   621→    if (dest) usedDestinations.add(dest.id);
   622→
   623→    const costLevel = budget === "luxury" ? "$$$$" : budget === "moderate" ? "$$" : "$";
   624→    const mealCost = budget === "luxury" ? "$$$" : budget === "moderate" ? "$$" : "$";
   625→
   626→    dayPlans.push({
   627→      day: dayNum,
   628→      title: dayNum === 1 ? "Arrival & First Impressions" : dayNum === days ? "Farewell & Departure" : `Exploring Day ${dayNum}`,
   629→      activities: [
   630→        {
   631→          time: "7:00 AM",
   632→          activity: "Breakfast at local restaurant",
   633→          location: dest ? `${dest.municipality}` : "Ho",
   634→          cost: mealCost,
   635→        },
   636→        {
   637→          time: "9:00 AM",
   638→          activity: `Visit ${dest ? dest.name : "Volta Lake"} and explore`,
   639→          location: dest ? dest.name : "Lake Volta",
   640→          cost: costLevel,
   641→        },
   642→        {
   643→          time: "1:00 PM",
   644→          activity: "Lunch with local cuisine",
   645→          location: dest ? dest.municipality : "Ho",
   646→          cost: mealCost,
   647→        },
   648→        {
   649→          time: "3:00 PM",
   650→          activity: dayNum <= 2
   651→            ? "Guided community tour & cultural immersion"
   652→            : dayNum === days
   653→              ? "Souvenir shopping & departure preparations"
   654→              : `Adventure activity near ${dest ? dest.name : "the site"}`,
   655→          location: dest ? dest.municipality : "Volta Region",
   656→          cost: costLevel,
   657→        },
   658→        {
   659→          time: "7:00 PM",
   660→          activity: "Dinner & evening relaxation",
   661→          location: dest ? dest.municipality : "Ho",
   662→          cost: mealCost,
   663→        },
   664→      ],
   665→    });
   666→  }
   667→
   668→  return { days, budget, interests, style, itinerary: dayPlans };
   669→}
