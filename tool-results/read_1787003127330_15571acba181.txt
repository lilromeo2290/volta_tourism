   146→   146→  const [toCurrency, setToCurrency] = useState("GHS");

   148→   148→  const [convertedAmount, setConvertedAmount] = useState<string | null>(null);
   149→   149→
   150→   150→  useEffect(() => {
   151→   151→    const fetchRates = async () => {
   152→   152→      try {
   153→   153→        const res = await fetch(
   154→   154→          "https://api.frankfurter.app/latest?from=GHS&to=USD,EUR,GBP,NGN,CNY"
   155→   155→        );
   156→   156→        const data = await res.json();
   157→   157→        // Store GHS->X rates, then invert so we have 1 X = ? GHS
   158→   158→        const ghstoOther: Record<string, number> = {};
   159→   159→        for (const [cur, rate] of Object.entries(data.rates)) {
   160→   160→          ghstoOther[cur] = rate as number;
   161→   161→        }
   162→   162→        // 1 GHS = 1/USD_rate USD, etc.
   163→   163→        const map: Record<string, number> = { GHS: 1 };
   164→   164→        for (const cur of CURRENCIES) {
   165→   165→          if (cur !== "GHS" && ghstoOther[cur]) {
   166→   166→            map[cur] = 1 / ghstoOther[cur]; // how much of this currency per 1 GHS
   167→   167→          }
   168→   168→        }
   169→   169→        setRatesMap(map);
   170→   170→      } catch {
   171→   171→        setRatesMap({ GHS: 1, USD: 0.0647, EUR: 0.0595, GBP: 0.0510, NGN: 103.5, CNY: 0.466 });

   174→   174→      }
