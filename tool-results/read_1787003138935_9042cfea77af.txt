   141→function ForexWidget() {
   142→  const [ratesMap, setRatesMap] = useState<Record<string, number>>({});
   143→  const [loading, setLoading] = useState(true);
   144→  const [mode, setMode] = useState<"rates" | "convert">("rates");
   145→  const [fromCurrency, setFromCurrency] = useState("USD");
   146→  const [toCurrency, setToCurrency] = useState("GHS");
   147→  const [amount, setAmount] = useState("");
   148→  const [convertedAmount, setConvertedAmount] = useState<string | null>(null);
   149→
   150→  useEffect(() => {
   151→    const fetchRates = async () => {
   152→      try {
   153→        const res = await fetch(
   154→          "https://api.frankfurter.app/latest?from=GHS&to=USD,EUR,GBP,NGN,CNY"
   155→        );
   156→        const data = await res.json();
   157→        // Store GHS->X rates, then invert so we have 1 X = ? GHS
   158→        const ghstoOther: Record<string, number> = {};
   159→        for (const [cur, rate] of Object.entries(data.rates)) {
   160→          ghstoOther[cur] = rate as number;
   161→        }
   162→        // 1 GHS = 1/USD_rate USD, etc.
   163→        const map: Record<string, number> = { GHS: 1 };
   164→        for (const cur of CURRENCIES) {
   165→          if (cur !== "GHS" && ghstoOther[cur]) {
   166→            map[cur] = 1 / ghstoOther[cur]; // how much of this currency per 1 GHS
   167→          }
   168→        }
   169→        setRatesMap(map);
   170→      } catch {
   171→        setRatesMap({ GHS: 1, USD: 0.0647, EUR: 0.0595, GBP: 0.0510, NGN: 103.5, CNY: 0.466 });
   172→      } finally {
   173→        setLoading(false);
   174→      }
   175→    };
   176→    fetchRates();
   177→    const interval = setInterval(fetchRates, 600000);
   178→    return () => clearInterval(interval);
   179→  }, []);
   180→
   181→  const convertCurrency = (from: string, to: string, value: number): number => {
   182→    if (!ratesMap[from] || !ratesMap[to]) return 0;
   183→    // Convert value in 'from' to GHS first, then to 'to'
   184→    const inGHS = value / ratesMap[from];
   185→    return inGHS * ratesMap[to];
   186→  };
   187→
   188→  useEffect(() => {
   189→    if (mode === "convert" && amount && !isNaN(Number(amount))) {
   190→      const result = convertCurrency(fromCurrency, toCurrency, Number(amount));
   191→      setConvertedAmount(result.toFixed(2));
   192→    } else {
   193→      setConvertedAmount(null);
   194→    }
   195→  }, [amount, fromCurrency, toCurrency, mode, ratesMap]);
   196→
   197→  const swapCurrencies = () => {
   198→    setFromCurrency(toCurrency);
   199→    setToCurrency(fromCurrency);
   200→    setAmount(convertedAmount || "");
   201→  };
   202→
   203→  const ratesList = CURRENCIES.filter((c) => c !== "GHS").map((currency) => {
   204→    const rate = ratesMap[currency] ? 1 / ratesMap[currency] : 0;
   205→    return { currency, rate, flag: CURRENCY_FLAGS[currency] || "\uD83C\uDFDB\uFE0F" };
   206→  });
   207→

   209→    return (
   210→      <div className="bg-white/10 backdrop-blur-md border border-white/20 rounded-2xl p-5 min-w-[280px] animate-pulse">

   220→      </div>
   221→    );
   222→  }
   223→
   224→  return (
   225→    <div className="bg-white/10 backdrop-blur-md border border-white/20 rounded-2xl p-5 min-w-[280px]">
   226→      {/* Header with tab toggle */}
   227→      <div className="flex items-center justify-between mb-3">

   237→        </button>
   238→      </div>
   239→
   240→      {mode === "rates" ? (
   241→        <>
   242→          <p className="text-white/50 text-xs mb-3">1 GHS equals</p>
